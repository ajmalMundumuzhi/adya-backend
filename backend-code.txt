const service = require("./users.service");

function handleError(res, error) {
  const status = error.statusCode ||
    (error.name === "ZodError" || error.name === "ValidationError" ? 400 :
    error.name === "CastError" ? 400 :
    error.code === 11000 ? 409 : 500);

  return res.status(status).json({
    success: false,
    message: status === 500 ? "Internal server error" : error.message
  });
}

async function register(req, res) {
  try {
    const data = registerSchema.parse(req.body);
    const result = await service.register(data);
    return res.status(201).json({ success: true, ...result });
  } catch (error) {
    return handleError(res, error);
  }
}

  try {
    const data = loginSchema.parse(req.body);
    const result = await service.login(data.email, data.password);

    return handleError(res, error);
  }

========== FILE: src/modules/users/users.service.js ==========


const mongoose = require("mongoose");
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");
const repository = require("./users.repository");

function makeError(message, statusCode) {
  return Object.assign(new Error(message), { statusCode });
}

function createToken(user) {
  if (!process.env.JWT_SECRET || process.env.JWT_SECRET.length < 32) {
    throw makeError("Authentication configuration error", 500);
  }

  return jwt.sign(
    { id: user._id.toString(), role: "customer" },
    process.env.JWT_SECRET,
    { expiresIn: "8h" }
  );
}

function safeUser(user) {
  return {
    id: user._id,
    name: user.name,
    email: user.email,
    phone: user.phone || "",
    emailVerifiedAt: user.emailVerifiedAt,
    createdAt: user.createdAt
  };
}

async function register(data) {
  const existing = await repository.findByEmail(data.email);

  if (existing) {
    throw makeError("Email already registered", 409);
  }

  const password = await bcrypt.hash(data.password, 12);

  const user = await repository.create({
    name: data.name,
    email: data.email,
    phone: data.phone || "",
    password
  });

  return {
    token: createToken(user),
    user: safeUser(user)
  };
}

async function login(email, password) {
  const user = await repository.findByEmail(email);

  if (!user || !user.isActive) {
    throw makeError("Invalid email or password", 401);
  }

  const validPassword = await bcrypt.compare(password, user.password);

  if (!validPassword) {
    throw makeError("Invalid email or password", 401);
  }

  return {
    token: createToken(user),
    user: safeUser(user)
  };
}

async function listUsers() {
  return repository.findAll();
}

async function getUser(id) {
  if (!mongoose.isValidObjectId(id)) {
    throw makeError("Invalid user ID", 400);
  }

  const user = await repository.findById(id);

  if (!user) {
    throw makeError("User not found", 404);
  }

  return user;
}

async function updateProfile(id, data) {
  const user = await getUser(id);

  if (data.email && data.email !== user.email) {
    const existing = await repository.findByEmail(data.email);

    if (existing) {
      throw makeError("Email already registered", 409);
    }
  }

  const updated = await repository.update(id, data);

  if (!updated) {
    throw makeError("User not found", 404);
  }

  return updated;
}

async function updateStatus(id, isActive) {
  await getUser(id);

  const updated = await repository.update(id, { isActive });

  if (!updated) {
    throw makeError("User not found", 404);
  }

  return updated;
}

module.exports = {
  register,
  login,
  listUsers,
  getUser,
  updateProfile,
  updateStatus
};

const { z } = require("zod");

const phoneSchema = z.string().trim().max(20).optional().or(z.literal(""));

const registerSchema = z.object({
  name: z.string().trim().min(2).max(100),
  email: z.string().trim().email().transform(v => v.toLowerCase()),
  phone: phoneSchema,
  password: z.string().min(8).max(72)
});

const loginSchema = z.object({
  email: z.string().trim().email().transform(v => v.toLowerCase()),
  password: z.string().min(1).max(72)
});

const updateProfileSchema = z.object({
  name: z.string().trim().min(2).max(100).optional(),
  email: z.string().trim().email().transform(v => v.toLowerCase()).optional(),
  phone: phoneSchema
}).refine(data => Object.keys(data).length > 0, {
  message: "Provide at least one field to update"
});

const updateUserStatusSchema = z.object({
  isActive: z.boolean()
});

module.exports = {
  registerSchema,
  loginSchema,
  updateProfileSchema,
  updateUserStatusSchema
};

const mongoose = require("mongoose");
const Product = require("../../models/Product");
const ProductVariant = require("../../models/ProductVariant");
const repository = require("./wishlist.repository");

function makeError(message, statusCode) {
  return Object.assign(new Error(message), { statusCode });
}

function validateId(id, label) {
  if (!mongoose.isValidObjectId(id)) {
    throw makeError(`Invalid ${label}`, 400);
  }
}

async function getWishlist(userId) {
  return repository.findByUser(userId);
}

async function addToWishlist(userId, data) {
  const { productId, variantId, sizeOptionId } = data;

  validateId(productId, "product ID");
  validateId(variantId, "variant ID");
  validateId(sizeOptionId, "size option ID");

  const product = await Product.findOne({
    _id: productId,
    isActive: true
  }).lean();

  if (!product) {
    throw makeError("Product not found or unavailable", 404);
  }

  const variant = await ProductVariant.findOne({
    _id: variantId,
    productId,
    isActive: true
  }).lean();

  if (!variant) {
    throw makeError("Product colour variant is unavailable", 404);
  }

  const size = variant.sizes.find(
    (item) =>
      item._id.toString() === sizeOptionId &&
      item.isActive !== false
  );

  if (!size) {
    throw makeError("Selected size is unavailable", 404);
  }

  return repository.add({
    userId,
    productId,
    variantId,
    sizeOptionId
  });
}

async function removeFromWishlist(userId, wishlistItemId) {
  validateId(wishlistItemId, "wishlist item ID");

  const result = await repository.remove({
    _id: wishlistItemId,
    userId
  });

  if (!result.deletedCount) {
    throw makeError("Wishlist item not found", 404);
  }

  return true;
}

module.exports = {
  getWishlist,
  addToWishlist,
  removeFromWishlist
};

  add,
  remove

========== FILE: src/middleware/auth.js ==========


const jwt = require("jsonwebtoken");
const User = require("../models/User");

function readToken(req, res) {
  const header = req.headers.authorization;

  if (!header || !header.startsWith("Bearer ")) {
    res.status(401).json({
      success: false,
      message: "Authentication required"
    });
    return null;
  }

  try {
    return jwt.verify(
      header.slice(7),
      process.env.JWT_SECRET
    );
  } catch {
    res.status(401).json({
      success: false,
      message: "Invalid or expired token"
    });
    return null;
  }
}

function requireAdmin(req, res, next) {
  const decoded = readToken(req, res);

  if (!decoded) return;

  if (decoded.role !== "admin") {
    return res.status(403).json({
      success: false,
      message: "Admin access required"
    });
  }

  req.admin = decoded;
  return next();
}

async function authenticateCustomer(req, res) {
  const decoded = readToken(req, res);

  if (!decoded) return false;

  if (decoded.role !== "customer") {
    res.status(403).json({
      success: false,
      message: "Customer access required"
    });
    return false;
  }

  try {
    const user = await User.findById(decoded.id)
      .select("_id isActive");

    if (!user || !user.isActive) {
      res.status(401).json({
        success: false,
        message: "Customer account is unavailable"
      });
      return false;
    }

    req.user = {
      id: user._id.toString(),
      role: "customer"
    };

    return true;
  } catch {
    res.status(500).json({
      success: false,
      message: "Internal server error"
    });
    return false;
  }
}

async function requireCustomer(req, res, next) {
  const authenticated = await authenticateCustomer(req, res);

  if (!authenticated) return;

  return next();
}

async function optionalCustomer(req, res, next) {
  const header = req.headers.authorization;

  if (!header) {
    return next();
  }

  if (!header.startsWith("Bearer ")) {
    return res.status(401).json({
      success: false,
      message: "Invalid authorization header"
    });
  }

  const authenticated = await authenticateCustomer(req, res);

  if (!authenticated) return;

  return next();
}

module.exports = {
  requireAdmin,
  requireCustomer,
  optionalCustomer
};
